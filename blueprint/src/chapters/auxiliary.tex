\chapter{Auxiliary processes and martingales}
\label{chap:auxiliary}

The analysis of the $\aRTS$ family rests on a handful of auxiliary stochastic
processes built from the assignments and responses. This chapter collects their
definitions and elementary (mostly martingale) properties. It corresponds to the
setup of the proof sketch (Section~3.2) and of Appendix~A.

\section{Conditional probabilities and the assignment martingale}
\label{sec:aux_assignment}

\begin{definition}[Conditional selection probability]\label{def:cond_prob}
\lean{AlphaRAR.aRTSSelProb}\leanok
\uses{def:assignment, def:filtration}
For $k \in [K]$ and $m \ge 1$,
\[
  p_{m,k} := \E[X_{m,k} \mid \filt_{m-1}].
\]
For a RAR procedure these coincide with the allocation probabilities: with the
indexing convention of Definition~\ref{def:rar}, $p_{m,k}$ is the $k$-th
component of the vector used to draw $X_m$.
\end{definition}

\begin{definition}[Assignment martingale]\label{def:M}
\lean{AlphaRAR.assignMG}\leanok
\uses{def:cond_prob}
For $k \in [K]$ set $\Delta M_{m,k} := X_{m,k} - p_{m,k}$ and
\[
  M_{n,k} := \sum_{m=1}^n (X_{m,k} - p_{m,k}), \qquad M_n := (M_{n,1},\dots,M_{n,K}).
\]
\end{definition}

\begin{lemma}[$M$ is a martingale]\label{lem:M_martingale}
\lean{AlphaRAR.martingale_assignMart}\leanok
\uses{def:M, def:filtration}
For each $k \in [K]$, $(M_{n,k})_{n\ge0}$ is an $(\filt_n)$-martingale with
$M_{0,k}=0$ and bounded increments $|\Delta M_{m,k}| \le 1$.
\end{lemma}

\begin{proof}\leanok
\uses{def:cond_prob}
$\E[\Delta M_{m,k}\mid\filt_{m-1}] = \E[X_{m,k}\mid\filt_{m-1}] - p_{m,k} = 0$
by Definition~\ref{def:cond_prob}, and $X_{m,k},p_{m,k}\in[0,1]$.
\end{proof}

\begin{lemma}[The assignment martingale is $O_p(\sqrt n)$]\label{lem:M_Op}
\lean{AlphaRAR.isBigOpOne_assignMart_div_sqrt}\leanok
\uses{def:M, lem:M_martingale}
For each $k \in [K]$, $M_{n,k} = O_p(\sqrt n)$.
\end{lemma}

\begin{proof}\leanok
\uses{lem:mart_bdd_incr_Op}
$M_{\cdot,k}$ is a martingale with $M_{0,k}=0$ and increments bounded by $1$
(Lemma~\ref{lem:M_martingale}), so Lemma~\ref{lem:mart_bdd_incr_Op} applies with
$c = 1$.
\end{proof}

\begin{lemma}[Count decomposition]\label{lem:count_decomp}
\lean{AlphaRAR.count_eq}\leanok
\uses{def:M, def:counts}
For all $n\ge1$ and $k\in[K]$,
$\displaystyle N_{n,k} = \sum_{m=1}^n p_{m,k} + M_{n,k}$.
\end{lemma}

\begin{proof}\leanok
Sum the identity $X_{m,k} = p_{m,k} + \Delta M_{m,k}$ over $m \le n$.
\end{proof}

\section{The response martingale}
\label{sec:aux_response}

\begin{definition}[Centered response martingale]\label{def:Q}
\lean{AlphaRAR.respMG, AlphaRAR.respMart}\leanok
\uses{def:assignment, def:responses}
For $k \in [K]$ set $Y_{m,k} := \xi_{m,k} - \theta_k$ and
\[
  Q_{n,k} := \sum_{m=1}^n X_{m,k}\,(\xi_{m,k} - \theta_k)
           = \sum_{m=1}^n X_{m,k}\,Y_{m,k},
  \qquad Q_n := (Q_{n,1}, \dots, Q_{n,K}).
\]
\end{definition}

\begin{lemma}[$Q$ is a martingale]\label{lem:Q_martingale}
\lean{AlphaRAR.martingale_respMart}\leanok
\uses{def:Q, def:filtration}
For each $k \in [K]$, if the responses are integrable then $(Q_{n,k})_{n\ge0}$ is a
$(\mathcal{G}_n)$-martingale, where $\mathcal{G}_n := \filt_{n-1}\vee\sigma(A_n)$ is the
action-augmented filtration (\texttt{filtrationAction}); a fortiori it is an
$(\filt_n)$-martingale.
\end{lemma}

\begin{proof}\leanok
Write $\Delta Q_{n,k} = X_{n,k}(\xi_{n,k} - \theta_k)$, where $X_{n,k}$ is the
indicator that the assignment $A_n$ equals arm $k$. The natural filtration is the
\emph{action-augmented} one $\mathcal{G}_n = \filt_{n-1}\vee\sigma(A_n)$
(\texttt{filtrationAction}): the assignment $X_{n,k}$ is $\mathcal{G}_n$-measurable
(the arm is known) while the response $\xi_{n,k}$ is not (it is revealed afterwards).
Conditioning on $\mathcal{G}_n$, the response is drawn from the arm's reward kernel
independently of the past given the arm, so its conditional mean is $\theta_k$;
pulling out the $\mathcal{G}_n$-measurable factor $X_{n,k}$ gives
$\E[\Delta Q_{n,k}\mid\mathcal{G}_n] = X_{n,k}(\theta_k-\theta_k)=0$, so $Q_{\cdot,k}$ is
a $\mathcal{G}$-martingale. Since $Q_{n,k}$ is $\filt_{n-1}\subseteq\mathcal{G}_n$-measurable,
the tower property also gives $\E[\Delta Q_{n,k}\mid\filt_{n-1}] = 0$, the paper's
$\filt$-martingale statement.

This is formalized in the \texttt{LeanMachineLearning} algorithm--environment
framework: the arm is the action and the response is the feedback of a stationary
environment with per-arm reward kernel $\nu$, and $\theta_k = \int x\,\nu_k(dx)$ is
the kernel mean. The conditional-mean computation is \texttt{condExp\_feedback}
(which handles the first patient and the general step via the two conditional-distribution
fields of the framework), and the filtration $\mathcal{G}$ is \texttt{filtrationAction}.
\end{proof}

\begin{lemma}[Conditional second moment of $\Delta Q$]\label{lem:Q_cond_second_moment}
\lean{AlphaRAR.condExp_respMart_increment_sq}\leanok
\uses{def:Q, def:condA}
For each $k \in [K]$, conditioning on the action-augmented $\sigma$-algebra
$\mathcal{G}_n = \filt_{n-1}\vee\sigma(A_n)$ (\texttt{filtrationAction}),
\[
  \E\big[(\Delta Q_{n,k})^2 \mid \mathcal{G}_n\big] = X_{n,k}\, V_k,
\]
where $V_k = \Var[\xi_{1,k}] = \int (x-\theta_k)^2\,\nu_k(dx)$ is the variance of arm $k$
(Mathlib's \texttt{variance}: \texttt{Var[id; ν k]}).
\end{lemma}

\begin{proof}\leanok
Since $X_{n,k}$ is a $\{0,1\}$-indicator, $X_{n,k}^2 = X_{n,k}$ and hence
$(\Delta Q_{n,k})^2 = X_{n,k}(\xi_{n,k}-\theta_k)^2$. The factor $X_{n,k}$ is
$\mathcal{G}_n$-measurable, so it pulls out of the conditional expectation; by the same
conditional-distribution computation as in Lemma~\ref{lem:Q_martingale} --- now applied to
$g(x) = (x-\theta_k)^2$ (\texttt{condExp\_feedback\_comp}) --- the response's conditional
second central moment given $\mathcal{G}_n$ is $\int (x-\theta_k)^2\,\nu_{A_n}(dx)$. On the
event $\{A_n = k\}$ this integral equals $V_k$, while off it $X_{n,k}=0$; in either case
the product is $X_{n,k} V_k$.
\end{proof}

\begin{lemma}[Quadratic variation of $Q$]\label{lem:Q_quad_var}
\lean{AlphaRAR.predQuadVar_respMart_eq, AlphaRAR.martingale_sq_sub_predQuadVar_respMart}\leanok
\uses{def:Q, def:counts, def:condA, def:pred_qv}
Under Condition~\textbf{A}, the predictable quadratic variation of $Q_{\cdot,k}$ is
$\langle Q_{\cdot,k}\rangle_n = V_k N_{n,k}$: the process $Q_{\cdot,k}^2 - V_k N_{\cdot,k}$ is a
martingale, with $V_k N_{\cdot,k}$ its predictable compensator.
\end{lemma}

\begin{proof}\leanok
\uses{lem:Q_cond_second_moment, lem:qv_mart}
Since $Q_{\cdot,k}$ is a martingale for the action-augmented filtration
$\mathcal{G} = \texttt{filtrationAction}$ (Lemma~\ref{lem:Q_martingale}), the paper's
$V_k N_{n,k}$ is the \emph{ordinary} predictable quadratic variation of $Q_{\cdot,k}$ for
$\mathcal{G}$ (Definition~\ref{def:pred_qv}): its increment is
$\E[(\Delta Q_{m+1,k})^2\mid\mathcal{G}_m] = V_k X_{m,k}$
(Lemma~\ref{lem:Q_cond_second_moment}), and summing over $m<n$ gives
$V_k\sum_{m<n} X_{m,k} = V_k N_{n,k}$ --- the indicator survives the sum because the current
arm $A_m$ is $\mathcal{G}_m$-measurable. That $Q^2-\langle Q\rangle$ is a martingale (for
$\mathcal{G}$) is Lemma~\ref{lem:qv_mart}. The sole hypothesis is finite arm variance
(Condition~\textbf{A}), formalized as $\xi_{\cdot,k}\in L^2$.
\end{proof}

\begin{lemma}[Second moment of $Q$]\label{lem:Q_second_moment}
\lean{AlphaRAR.integral_respMart_sq_eq}\leanok
\uses{def:Q, def:counts, lem:Q_quad_var}
Under Condition~\textbf{A}, for each $k \in [K]$,
$\E[Q_{n,k}^2] = V_k\,\E[N_{n,k}]$.
\end{lemma}

\begin{proof}\leanok
\uses{lem:qv_second_moment}
The discrete Itô isometry (Lemma~\ref{lem:qv_second_moment}) gives
$\E[Q_{n,k}^2] = \E[\langle Q_{\cdot,k}\rangle_n]$ for the martingale $Q_{\cdot,k}$ (which starts at
$0$), and $\langle Q_{\cdot,k}\rangle_n = V_k N_{n,k}$ by Lemma~\ref{lem:Q_quad_var}; taking
expectations, $\E[Q_{n,k}^2] = V_k\,\E[N_{n,k}]$.
\end{proof}

\begin{lemma}[Cross variation of $Q$]\label{lem:Q_cross_var}
\lean{AlphaRAR.martingale_respMart_mul}\leanok
\uses{def:Q, def:counts}
For $k \ne j$, the cross variation vanishes: $\langle Q_{\cdot,k}, Q_{\cdot,j}\rangle_n = 0$.
Equivalently, $Q_{\cdot,k}\,Q_{\cdot,j}$ is a martingale (for the action-augmented filtration
$\mathcal{G}$).
\end{lemma}

\begin{proof}\leanok
\uses{lem:qv_orthogonal}
The increments $\Delta Q_{\cdot,k}$ and $\Delta Q_{\cdot,j}$ are orthogonal --- indeed their
product is \emph{identically} zero, since $X_{m,k}X_{m,j} = 0$ for $k \ne j$ (each patient is
assigned to exactly one arm, \texttt{respMart\_increment\_mul\_eq\_zero}). Hence
$Q_{\cdot,k}\,Q_{\cdot,j}$ is a martingale by Lemma~\ref{lem:qv_orthogonal}
(\texttt{martingale\_mul}), so its predictable compensator --- the cross variation --- is $0$.
\end{proof}

\begin{lemma}[The response martingale is $O_p(\sqrt n)$]\label{lem:Q_Op}
\lean{AlphaRAR.isBigOpOne_respMart_div_sqrt}\leanok
\uses{def:Q, lem:Q_martingale}
For each $k \in [K]$, if the responses are square-integrable (Condition~\textbf{A}), then
$Q_{n,k} = O_p(\sqrt n)$.
\end{lemma}

\begin{proof}\leanok
\uses{cor:mart_Op, lem:Q_cond_second_moment}
$Q_{\cdot,k}$ is a martingale (for the action-augmented filtration $\mathcal{G}$) with
$Q_{0,k}=0$, and its increment second moments are bounded by the arm variance: taking
expectations in Lemma~\ref{lem:Q_cond_second_moment} via the tower property,
$\E[(\Delta Q_{n+1,k})^2] = \E[X_{n,k}\,V_k] = V_k\,\Prob(X_{n,k}=1) \le V_k$
(\texttt{integral\_respMart\_increment\_sq\_le}). So Corollary~\ref{cor:mart_Op} applies with
$\sigma^2 = V_k$.
\end{proof}

\begin{lemma}[Estimator error via $Q$]\label{lem:theta_error_Q}
\lean{AlphaRAR.theta_error_Q}\leanok
\uses{def:estimator, def:Q, def:bahadur}
On $\{N_{m,k}\ge1\}$, the leading term of the estimator error is
$\dfrac{1}{N_{m,k}}\sum_{j=1}^m X_{j,k}\xi_{j,k} - \theta_k
 = \dfrac{Q_{m,k}}{N_{m,k}}$.
Consequently, under the Bahadur representation
(Definition~\ref{def:bahadur}),
$\hat{\theta}_{m,k} - \theta_k = \dfrac{Q_{m,k}}{N_{m,k}} + o(N_{m,k}^{-1/2})$.
\end{lemma}

\begin{proof}
$\frac{1}{N_{m,k}}\sum_j X_{j,k}\xi_{j,k} - \theta_k
= \frac{1}{N_{m,k}}\sum_j X_{j,k}(\xi_{j,k}-\theta_k)
= \frac{Q_{m,k}}{N_{m,k}}$, using $\sum_j X_{j,k} = N_{m,k}$.
\leanok
\end{proof}

\section{The auxiliary process \texorpdfstring{$U$}{U} and the last hitting time}
\label{sec:aux_U}

\begin{definition}[Auxiliary process $U$]\label{def:U}
\lean{AlphaRAR.auxU}\leanok
\uses{def:M, def:plugin_target}
For a fixed $\alpha \in [0,1)$, $k \in [K]$ and $n \ge 0$ (with the $0$-indexed convention,
patient $0$ first),
\[
  U_{n,k} := \sum_{m=0}^{n-1} \alpha\, \hat{\rho}_{m,k} + M_{n,k} - n\, \hat{\rho}_{n,k}.
\]
The leading sum runs over $m < n$, so it uses the plug-in targets of patients $0,\dots,n-1$.
\end{definition}

\begin{lemma}[Increment of $U$]\label{lem:U_increment}
\lean{AlphaRAR.auxU_succ_sub}\leanok
\uses{def:U}
For all $n \ge 0$ and $k \in [K]$,
\[
  \Delta U_{n,k} = U_{n+1,k} - U_{n,k}
  = \alpha\, \hat{\rho}_{n,k} - p_{n,k} + \Delta\big(N_{n,k} - n\hat{\rho}_{n,k}\big),
\]
where we used $\Delta M_{n,k} = X_{n,k} - p_{n,k}$ and $\Delta N_{n,k} = X_{n,k}$. The selection
probability $p_{n,k}$ is paired with the target $\hat\rho_{n,k}$ at the \emph{same} index, so the
throttle discharged downstream is $p_{n,k} \le \alpha\hat\rho_{n,k}$ (patient $n$ is allocated using
the target of patients $0,\dots,n-1$). The identity holds at $n=0$ too, since the $\alpha\hat\rho$
sum grows at every step.
\end{lemma}

\begin{proof}\leanok
Direct computation from Definition~\ref{def:U}, expanding
$M_{n+1,k} - M_{n,k} = X_{n,k} - p_{n,k}$ and telescoping.
\end{proof}

\begin{definition}[Last under-sampling time]\label{def:hitting}
\lean{AlphaRAR.hitting}\leanok
\uses{def:counts, def:plugin_target}
For $k \in [K]$ and $n \ge 1$, the \emph{last under-sampling time} before $n$ is
\[
  \ell_{n,k} := \max\Big\{\, m \le n : \frac{N_{m,k}}{m} \le \hat{\rho}_{m,k} \,\Big\}.
\]
It is the last time up to $n$ at which arm $k$ is under-sampled (its proportion
is at most its estimated target).
\end{definition}

\begin{lemma}[Basic properties of the hitting time]\label{lem:hitting_basic}
\lean{AlphaRAR.hitting_basic}\leanok
\uses{def:hitting}
For each $k$, $(\ell_{n,k})_{n\ge1}$ is non-decreasing and $\ell_{n,k} \le n$.
\end{lemma}

\begin{proof}\leanok
Immediate from the definition: enlarging $n$ enlarges the set over which the max
is taken, and the max is bounded by $n$.
\end{proof}

\textbf{Lean remark.} Well-definedness of $\ell_{n,k}$ requires the set
$\{m\le n : N_{m,k}/m \le \hat\rho_{m,k}\}$ to be nonempty; a convenient
convention (used implicitly in the paper) is that it contains a base time such
as the end of the burn-in, or one sets $\ell_{n,k}=0$ when the set is empty.
The proofs only use Lemma~\ref{lem:hitting_basic} together with the sign
property of the next lemma.

\begin{lemma}[Sign at the hitting time]\label{lem:hitting_sign}
\lean{AlphaRAR.hitting_sign}\leanok
\uses{def:hitting, def:aRTS}
For $m > \ell_{n,k}$ (and $m \le n$), $\frac{N_{m,k}}{m} > \hat{\rho}_{m,k}$;
in particular, for an $\aRTS$ design with $m \ge Km_0$, $p_{m+1,k} \le \alpha\hat{\rho}_{m,k}$.
\end{lemma}

\begin{proof}\leanok
By maximality of $\ell_{n,k}$, no $m$ with $\ell_{n,k} < m \le n$ satisfies
$N_{m,k}/m \le \hat{\rho}_{m,k}$. The second statement is the throttling
condition \eqref{eq:throttle}.
\end{proof}
