\chapter{Strong law of large numbers for martingales}
\label{chap:pre_slln}

We formalize the $L^2$ strong law of large numbers for martingales: if the
increment variances decay fast enough (in particular for uniformly bounded
increments), then $M_n / n \to 0$ a.s. This is the result used in
Lemma~\ref{lem:M_lln} to control the assignment martingale.

\textbf{Formalization note.} Mathlib has the i.i.d.\ strong law
(\texttt{Probability/StrongLaw.lean}) but no martingale version. The classical
route below is Kolmogorov's: turn the claim into the a.s.\ convergence of a
series via the $L^2$ martingale convergence theorem (which Mathlib \emph{does}
have, \texttt{Probability/Martingale/Convergence.lean}), then apply Kronecker's
lemma. The only genuinely new pieces are Kronecker's lemma (elementary real
analysis) and the assembly.

\section{Ingredients}
\label{sec:pre_slln_ingredients}

\begin{lemma}[Orthogonality of martingale increments]\label{lem:slln_orthogonal}
Let $(M_n)$ be a square-integrable martingale. For $i \ne j$,
$\E[\Delta M_i\, \Delta M_j] = 0$; consequently
$\E[M_n^2] = \sum_{i=1}^n \E[(\Delta M_i)^2]$ when $M_0 = 0$.
\end{lemma}

\begin{proof}
For $i < j$, condition on $\filt_{j-1}$: $\E[\Delta M_i \Delta M_j]
= \E[\Delta M_i\, \E[\Delta M_j\mid\filt_{j-1}]] = 0$. Expanding
$M_n^2 = (\sum_i \Delta M_i)^2$ and using orthogonality kills the cross terms.
\end{proof}

\begin{lemma}[Weighted increment martingale is $L^2$-bounded]\label{lem:slln_weighted}
\lean{AlphaRAR.martingale_weightedSeries}
\uses{lem:slln_orthogonal}
Let $(M_n)$ be a square-integrable martingale with $M_0=0$ and
$\sum_{k\ge1}\frac{\E[(\Delta M_k)^2]}{k^2} < \infty$. Then
$S_n := \sum_{k=1}^n \frac{\Delta M_k}{k}$ is a square-integrable martingale with
$\sup_n \E[S_n^2] = \sum_{k\ge1}\frac{\E[(\Delta M_k)^2]}{k^2} < \infty$.
\end{lemma}

\begin{proof}
Each increment $\Delta S_k = \Delta M_k / k$ is $\filt_k$-measurable with
$\E[\Delta S_k\mid\filt_{k-1}] = 0$, so $S_n$ is a martingale. By
Lemma~\ref{lem:slln_orthogonal}, $\E[S_n^2] = \sum_{k\le n}\E[(\Delta M_k)^2]/k^2$,
which is bounded by the convergent series.
\end{proof}

\begin{lemma}[$L^2$-bounded martingales converge a.s.]\label{lem:slln_l2_conv}
\lean{AlphaRAR.martingale_ae_tendsto_of_eLpNorm_two_le}\leanok
A martingale $(S_n)$ with $\sup_n \|S_n\|_{L^2} < \infty$ converges a.s.\ to a finite limit.
\end{lemma}

\begin{proof}\leanok
On a probability space $\|\cdot\|_{L^1}\le\|\cdot\|_{L^2}$
(\texttt{eLpNorm\_le\_eLpNorm\_of\_exponent\_le}), so the uniform $L^2$ bound gives a uniform
$L^1$ bound, and $(S_n)$ is an $L^1$-bounded (sub)martingale. The $L^1$ martingale convergence
theorem (Mathlib's \texttt{Submartingale.exists\_ae\_tendsto\_of\_bdd}) then gives a.s.\ convergence
to a finite limit.
\end{proof}

\begin{lemma}[Kronecker's lemma, identity weights]\label{lem:kronecker}
\lean{AlphaRAR.kronecker}\leanok
Let $(x_n)$ be reals such that the partial sums $\sum_{k<n} x_k$ converge (to some $s$). Then
$\frac{1}{n}\sum_{k<n} k\,x_k \to 0$.
\end{lemma}

\begin{proof}\leanok
This is the special case $b_k = k$ of Kronecker's lemma, which is the only case used below
(Theorem~\ref{thm:mart_slln}). Write $s_n = \sum_{k<n} x_k \to s$. Summation by parts
(\texttt{Finset.sum\_range\_by\_parts}) gives
$\sum_{k<n} k\,x_k = (n-1)s_n - \sum_{k<n} s_k$, so
$\frac{1}{n}\sum_{k<n} k\,x_k = \big(1-\tfrac1n\big)s_n - \frac{1}{n}\sum_{k<n} s_k$.
The first term tends to $s$ and the second is a Cesàro average of the convergent sequence $(s_n)$,
hence also tends to $s$ (\texttt{Filter.Tendsto.cesaro}); the difference tends to $0$.
\end{proof}

\textbf{Formalization note.} Kronecker's lemma is not in Mathlib (only unrelated
``Kronecker'' matrix results exist). The identity-weight case above is a short, self-contained
real-analysis lemma; the general positive-nondecreasing-weight version
(Lemma~\ref{lem:kronecker_general}) is a natural standalone contribution, needed for the
bracket-normalized strong law below.

\section{Bracket-normalized strong law}
\label{sec:pre_slln_bracket}

The $n$-normalized law $M_n/n\to0$ (Theorem~\ref{thm:mart_slln}) is not enough for the estimator
dichotomy (Lemma~\ref{lem:theta_limit_dichotomy}): there the martingale $Q_{\cdot,k}$ must be divided
by its own (random) sample size $N_{n,k}$, which can grow slower than $n$. What is needed is the
\emph{bracket-normalized} strong law $M_n/\langle M\rangle_n\to0$ on $\{\langle M\rangle_n\to\infty\}$
--- a general, reusable martingale theorem absent from Mathlib. Its proof is the martingale-transform
argument, and it consumes the general Kronecker lemma.

\begin{lemma}[Kronecker's lemma, general weights]\label{lem:kronecker_general}
\lean{AlphaRAR.kronecker_general}\leanok
Let $(b_n)$ be positive and nondecreasing with $b_n\to\infty$, and let $(y_n)$ be reals whose partial
sums $\sum_{k<n} y_k$ converge (to some $s$). Then $\dfrac1{b_n}\sum_{k<n} b_{k+1}\,y_k \to 0$.
\end{lemma}

\begin{proof}\leanok
Summation by parts: with $S_n=\sum_{k<n}y_k$,
$\sum_{k<n} b_{k+1}y_k = b_n S_n - \sum_{k<n-1}(b_{k+2}-b_{k+1})S_{k+1}$. Dividing by $b_n$, the first
term is $S_n\to s$; the second is a Toeplitz-weighted average of the convergent sequence
$(S_{k+1})$ with nonnegative weights $(b_{k+2}-b_{k+1})/b_n$ summing to $(b_n-b_1)/b_n\to1$, each
tending to $0$. Writing $S_{k+1}=s+\varepsilon_k$ with $\varepsilon_k\to0$ and splitting the sum at a
threshold beyond which $|\varepsilon_k|<\delta$, the weighted average tends to $s$; the difference
tends to $0$.
\end{proof}

\begin{lemma}[Bracket-weighted transform is $L^2$-bounded]\label{lem:slln_bracket_weighted}
\lean{AlphaRAR.martingale_bracketSeries, AlphaRAR.predQuadVar_bracketSeries_le_one}\leanok
\uses{def:pred_qv}
For a square-integrable martingale $(M_n)$ with $M_0=0$, the transform
\[
  T_n := \sum_{k<n} \frac{\Delta M_k}{1+\langle M\rangle_{k+1}}
\]
is a martingale (the weights $1/(1+\langle M\rangle_{k+1})$ are $\filt_k$-measurable, as
$\langle M\rangle$ is predictable), and it is $L^2$-bounded:
$\E[T_n^2] = \sum_{k<n} \E\!\big[\tfrac{\Delta\langle M\rangle_k}{(1+\langle M\rangle_{k+1})^2}\big]
\le \E\!\big[\tfrac1{1+\langle M\rangle_0}-\tfrac1{1+\langle M\rangle_n}\big] \le 1$,
using $\E[(\Delta M_k)^2\mid\filt_k]=\Delta\langle M\rangle_k$ and the telescoping bound
$\tfrac{\Delta\langle M\rangle_k}{(1+\langle M\rangle_{k+1})^2}\le
\tfrac1{1+\langle M\rangle_k}-\tfrac1{1+\langle M\rangle_{k+1}}$.
\end{lemma}

\begin{proof}\leanok
\uses{lem:qv_incr, lem:slln_orthogonal}
The martingale property and Itô isometry give
$\E[T_n^2]=\sum_{k<n}\E[(\Delta T_k)^2]=\sum_{k<n}\E[(\Delta M_k)^2/(1+\langle M\rangle_{k+1})^2]$;
pull the $\filt_k$-measurable weight out of the conditional expectation and use
$\E[(\Delta M_k)^2\mid\filt_k]=\Delta\langle M\rangle_k$ (Lemma~\ref{lem:qv_incr}). The telescoping
bound holds because $(1+\langle M\rangle_k)(1+\langle M\rangle_{k+1})\le(1+\langle M\rangle_{k+1})^2$.
\end{proof}

\begin{theorem}[Bracket-normalized martingale SLLN]\label{thm:mart_slln_bracket}
\lean{AlphaRAR.martingale_div_predQuadVar_ae_tendsto_zero}\leanok
\uses{lem:slln_bracket_weighted, lem:slln_l2_conv, lem:kronecker_general}
Let $(M_n)$ be a square-integrable martingale with $M_0=0$. Then almost surely on
$\{\langle M\rangle_n\to\infty\}$, $\dfrac{M_n}{\langle M\rangle_n}\to0$.
\end{theorem}

\begin{proof}\leanok
\uses{lem:slln_bracket_weighted, lem:slln_l2_conv, lem:kronecker_general}
By Lemma~\ref{lem:slln_bracket_weighted} the transform $T_n$ is an $L^2$-bounded martingale, so
$T_n\to T_\infty$ a.s.\ (Lemma~\ref{lem:slln_l2_conv}). Apply the general Kronecker lemma
(Lemma~\ref{lem:kronecker_general}) pathwise with $b_k=1+\langle M\rangle_k$ (nondecreasing, and
$\to\infty$ on the event) and $y_k=\Delta M_k/(1+\langle M\rangle_{k+1})=\Delta T_k$:
$\frac1{b_n}\sum_{k<n} b_{k+1}\,y_k=\frac{M_n}{1+\langle M\rangle_n}\to0$, hence
$M_n/\langle M\rangle_n\to0$.
\end{proof}

\section{The theorem}
\label{sec:pre_slln_thm}

\begin{theorem}[Martingale SLLN]\label{thm:mart_slln}
\lean{AlphaRAR.martingale_div_atTop_ae_tendsto_zero}\leanok
\uses{lem:slln_weighted, lem:slln_l2_conv, lem:kronecker}
Let $(M_n)$ be a square-integrable martingale with $M_0 = 0$ and
$\sum_{k\ge1}\frac{\E[(\Delta M_k)^2]}{k^2} < \infty$. Then $\frac{M_n}{n}\to 0$
a.s.
\end{theorem}

\begin{proof}\leanok
By Lemmas~\ref{lem:slln_weighted} and \ref{lem:slln_l2_conv}, the series
$S_n = \sum_{k\le n}\Delta M_k/k$ converges a.s. Apply Kronecker's lemma
(Lemma~\ref{lem:kronecker}) pathwise with $b_k = k$ and $x_k = \Delta M_k/k$:
$\frac{1}{n}\sum_{k\le n} k\cdot\frac{\Delta M_k}{k} = \frac{M_n}{n}\to 0$ a.s.
\end{proof}

\begin{corollary}[Bounded increments]\label{cor:mart_slln_bounded}
\lean{AlphaRAR.martingale_div_atTop_ae_tendsto_zero_of_bdd}\leanok
\uses{thm:mart_slln}
If $(M_n)$ is a martingale with $M_0=0$ and uniformly bounded increments
$|\Delta M_k| \le c$, then $\frac{M_n}{n}\to 0$ a.s.
\end{corollary}

\begin{proof}\leanok
$\E[(\Delta M_k)^2]\le c^2$, so $\sum_k \E[(\Delta M_k)^2]/k^2 \le c^2\sum_k k^{-2}
< \infty$; apply Theorem~\ref{thm:mart_slln}.
\end{proof}
